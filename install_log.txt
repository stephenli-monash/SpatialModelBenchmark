setwd('/fs04/scratch2/tc48/SPIAT/SpatialModelBenchmark/')
devtools::load_all()

params <- list(
  patient_col = "SampleID",
  test_column = "TissueRegion",

  Total_ROI_Area = "Area.mm2",
  Total_Cell_Number = "Total_Cell_Number",

  FOV_col = "FOV_ID",

  colWilcox = "dens",
  colGLMM = "_counts",

  GLMM_formula = "+ (1|patient_id) + offset(log(Area.mm2))"
)

glmm_models <- list(
   nbinom1 = list(
    family = "nbinom1",
    link = "log",
    formula = params$GLMM_formula
  ),
  nbinom2 = list(
    family = "nbinom2",
    link = "log",
    formula = params$GLMM_formula
  ),
  poisson = list(
    family = "poisson",
    link = "log",
    formula = params$GLMM_formula
  ),
 tweedie = list(
    family = "tweedie", 
    link = "log",
    formula = params$GLMM_formula
  )
)

run_spatial_benchmark(
  data_dir = "/fs04/scratch2/tc48/SPIAT/OPAL_MAC/extracted_features/",
  output_dir = "/fs04/scratch2/tc48/SPIAT/OPAL_MAC/html/",
  params = params,
  glmm_models = glmm_models,
   n_shuffle=100
)
